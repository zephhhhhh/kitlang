42/// Represents errors from different stages in the compilation process.
43#[derive(Debug, Error)]
44pub enum KitlangError {
45    /// Errors originating from the parser stage.
46    #[error("Parser Error: {0}")]
47    ParserError(#[from] crate::parser::ParseError),
48    /// Errors originating from the lowering stage.
49    #[error("Lowering Error: {0}")]
50    LoweringError(#[from] crate::intermediate::hir::LoweringError),
51    /// Indicates that execution ended unexpectedly, I.e. the interpreter encountered an error.
52    #[error("Execution ended unexpectedly.")]
53    ExecutionEndedUnexpectedly,
54    /// Indicates that the entry point 'main' function could not be found when trying to execute.
55    #[error("Failed to find entry point 'main' function.")]
56    FailedToFindEntryPoint,
57}
58
59impl KitlangError {
60    /// Format the error as an error message, given the source code string.
61    #[inline]
62    #[must_use]
63    pub fn format_as_error_message(&self, source_string: &str) -> String {
64        match self {
65            Self::ParserError(err) => err.format_as_error_message(source_string),
66            Self::LoweringError(err) => err.format_as_error_message(source_string),
67            e => SpannedErrorBuilder::new(source_string, SourceSpan::null_span())
68                .print_header_line(format!("{e}"))
69                .generate_highlight()
70                .generate_output(),
71        }
72    }
73}

78/// Parse a given kitlang source code string into MIR representations.
79/// This function injects the standard library, if you do not want that, use `parse_source_string_to_mir_no_std` instead.
80/// # Errors
81/// This function will return an error if any part of the parsing or lowering process fails.
82/// The returned error will contain a diagnostic message indicating the nature and location of any failures, as well
83/// as which stage they originate from.
84///
85/// These errors can be formatted into user-friendly error messages using the `format_as_error_message`
86/// method on the `KitlangError` on the `Err` variant.
87pub fn parse_source_string_to_mir(
88    source: &str,
89) -> KitlangResult<(intermediate::hir::ProgramMetaData, intermediate::mir::MIR)> {
90    let final_source = inject_standard_library(source);
91
92    let ast = crate::parser::parse_from_source(&final_source)?;
93    let (meta_data, hir) = intermediate::hir::parse_ast_to_hir_processed(&ast)?;
94    let mir = intermediate::mir::lower_hir_to_mir(&hir, &meta_data)?;
95    Ok((meta_data, mir))
96}

115/// Inject the standard library code into a provided source code string.
116#[inline]
117#[must_use]
118pub fn inject_standard_library(source: &str) -> String {
119    const CORE_LIB: &str = include_str!("../kit-std/core.purr");
120    const INTRINSICS_LIB: &str = include_str!("../kit-std/intrinsics.purr");
121
122    let mut source_with_std =
123        String::with_capacity(source.len() + CORE_LIB.len() + INTRINSICS_LIB.len() + 2);
124    source_with_std.push_str(source);
125    source_with_std.push('\n');
126    source_with_std.push_str(CORE_LIB);
127    source_with_std.push('\n');
128    source_with_std.push_str(INTRINSICS_LIB);
129
130    source_with_std
131}

133/// Execute a given kitlang source code string with the MIR interpreter, using the provided native functions.
134/// The standard library is injected into the program with this function, if you do not want this for some reason,
135/// use `execute_source_string_no_std` instead.
136/// If `time_execution` is `true`, the different stages of execution will be timed and printed
137/// # Errors
138/// This function will return an error if any part of the parsing, lowering or execution process fails.
139/// The returned error will contain a diagnostic message indicating the nature and location of any failures, as well
140/// as which stage they originate from.
141///
142/// These errors can be formatted into user-friendly error messages using the `format_as_error_message`
143/// method on the `KitlangError` on the `Err` variant.
144pub fn execute_source_string(
145    source: &str,
146    native_functions: crate::interpreter::mir_interpreter::RegisterNativeFns,
147    time_execution: bool,
148) -> KitlangResult<crate::interpreter::mir_interpreter::Value> {
149    use crate::compiler::Compiler;
150    use crate::interpreter::mir_interpreter::execute_mir;
151
152    let mut compiler = Compiler::new(source).profile_stages(time_execution);
153    let mir = compiler.compile_to_mir()?;
154
155    execute_mir(
156        mir,
157        compiler.context.meta(),
158        native_functions,
159        time_execution,
160    )
161}

163/// Execute a given kitlang source code string with the MIR interpreter, using the provided native functions.
164/// If `time_execution` is `true`, the different stages of execution will be timed and printed.
165/// # Errors
166/// This function will return an error if any part of the parsing, lowering or execution process fails.
167/// The returned error will contain a diagnostic message indicating the nature and location of any failures, as well
168/// as which stage they originate from.
169///
170/// These errors can be formatted into user-friendly error messages using the `format_as_error_message`
171/// method on the `KitlangError` on the `Err` variant.
172pub fn execute_source_string_no_std(
173    source: &str,
174    native_functions: crate::interpreter::mir_interpreter::RegisterNativeFns,
175    time_execution: bool,
176) -> KitlangResult<crate::interpreter::mir_interpreter::Value> {
177    use crate::compiler::Compiler;
178    use crate::interpreter::mir_interpreter::execute_mir_no_intrinsics;
179
180    let mut compiler = Compiler::new(source)
181        .profile_stages(time_execution)
182        .no_stdlib(true);
183    let mir = compiler.compile_to_mir()?;
184
185    execute_mir_no_intrinsics(
186        mir,
187        compiler.context.meta(),
188        native_functions,
189        time_execution,
190    )
191}

193/// Call this function to initialise logging, if the `logging` feature is enabled.
194pub fn init_logging() {
195    #[cfg(all(feature = "logging", not(feature = "webasm")))]
196    fn setup_logging() {
197        simplelog::CombinedLogger::init(vec![simplelog::TermLogger::new(
198            log::LevelFilter::Debug,
199            simplelog::Config::default(),
200            simplelog::TerminalMode::Mixed,
201            simplelog::ColorChoice::Auto,
202        )])
203        .expect("Failed to initialize logging");
204    }
205    #[cfg(all(feature = "logging", not(feature = "webasm")))]
206    setup_logging();
207}
