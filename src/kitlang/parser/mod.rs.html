2//!

34/// A cursor into a list of tokens, provides functionality for peeking as well as consuming tokens
35/// in a linear order.
36#[derive(Debug)]
37pub struct TokenCursor<'a> {
38    pub tokens: &'a TokenList,
39    pub position: u32,
40}
41
42impl<'a> TokenCursor<'a> {
43    #[inline]
44    #[must_use]
45    pub const fn new(tokens: &'a TokenList) -> Self {
46        Self {
47            tokens,
48            position: 0,
49        }
50    }
51}

54impl TokenCursor<'_> {
55    /// Returns the total number tokens in the underlying [`TokenList`].
56    #[allow(clippy::cast_possible_truncation)]
57    #[inline]
58    #[must_use]
59    pub const fn len(&self) -> u32 {
60        self.tokens.len() as u32
61    }
62
63    /// Returns if the total number of tokens in the underlying [`TokenList`] is `zero`.
64    #[inline]
65    #[must_use]
66    pub const fn is_empty(&self) -> bool {
67        self.tokens.is_empty()
68    }
69
70    /// Returns the current position of the [`TokenCursor`] as an index.
71    #[inline]
72    #[must_use]
73    pub const fn position(&self) -> u32 {
74        self.position
75    }
76
77    /// Returns the total number of tokens remaining until the end of the [`TokenList`].
78    #[inline]
79    #[must_use]
80    pub const fn remaining(&self) -> u32 {
81        self.len() - self.position
82    }
83
84    /// Returns true if the current cursor position is at the end of the [`TokenList`].
85    #[inline]
86    #[must_use]
87    pub const fn is_end(&self) -> bool {
88        self.position >= self.len()
89    }
90
91    /// Returns a reference to a [`Token`] at a specified absolute index into the [`TokenList`].
92    #[inline]
93    #[must_use]
94    pub fn get(&self, index: u32) -> Option<&Token> {
95        self.tokens.get(index as usize)
96    }
97
98    /// Returns the end of the current token.
99    #[inline]
100    #[must_use]
101    pub fn get_current_source_end(&self) -> u32 {
102        self.get(self.position()).map_or_else(
103            || self.tokens.last().map_or(0, |token| token.span.end),
104            |token| token.span.end,
105        )
106    }
107
108    /// Returns the start of the current token.
109    #[inline]
110    #[must_use]
111    pub fn get_current_source_start(&self) -> u32 {
112        self.get(self.position()).map_or_else(
113            || self.tokens.last().map_or(0, |token| token.span.start),
114            |token| token.span.start,
115        )
116    }
117
118    /// Returns the start of the previous token.
119    #[inline]
120    #[must_use]
121    pub fn get_previous_source_start(&self) -> u32 {
122        self.get(self.position().saturating_sub(1))
123            .map_or_else(|| self.get_current_source_start(), |token| token.span.start)
124    }
125
126    /// Returns the end of the previous token.
127    #[inline]
128    #[must_use]
129    pub fn get_previous_source_end(&self) -> u32 {
130        self.get(self.position().saturating_sub(1))
131            .map_or_else(|| self.get_current_source_end(), |token| token.span.end)
132    }
133
134    /// Returns a range that represents the start token.
135    #[inline]
136    #[must_use]
137    pub fn first_span(&self) -> Range<u32> {
138        self.tokens
139            .first()
140            .map_or(0..0, |token| token.span.start..token.span.end)
141    }
142
143    /// Returns a range that represents the end of the file.
144    #[inline]
145    #[must_use]
146    pub fn eof_span(&self) -> Range<u32> {
147        self.tokens
148            .last()
149            .map_or(0..0, |token| token.span.start..token.span.end)
150    }
151
152    /// Returns a range that represents the current token index.
153    #[inline]
154    #[must_use]
155    pub fn current_span(&self) -> Range<u32> {
156        self.get(self.position())
157            .map_or_else(|| self.eof_span(), |token| token.span.start..token.span.end)
158    }
159
160    /// Returns a range that represents the full file of tokens.
161    /// From the first character of the first token, to the last character of the final token.
162    #[inline]
163    #[must_use]
164    pub fn full_span(&self) -> Range<u32> {
165        let Some(first_token) = self.tokens.first() else {
166            return self.eof_span();
167        };
168        let Some(last_token) = self.tokens.last() else {
169            return self.eof_span();
170        };
171        (first_token.span.start)..(last_token.span.end)
172    }
173}

175impl TokenCursor<'_> {
176    /// Advance the token cursor by one position
177    /// # Returns
178    /// `true` if the cursor could advance, `false` otherwise.
179    #[inline]
180    pub const fn advance(&mut self) -> bool {
181        if self.position < self.len() {
182            self.position = self.position.saturating_add(1);
183            true
184        } else {
185            false
186        }
187    }
188
189    /// Advance the token cursor by `count` positions.
190    /// # Returns
191    /// The number of tokens the cursor was advanced by, may be less if reaching the end.
192    #[inline]
193    pub fn advance_by(&mut self, count: u32) -> u32 {
194        let adjusted = count.min(self.remaining());
195        self.position = self.position.saturating_add(adjusted);
196        adjusted
197    }
198
199    /// Get the current token and advance the cursor by one position.
200    #[inline]
201    pub fn consume(&mut self) -> Option<&Token> {
202        if self.position >= self.len() {
203            None
204        } else {
205            let position = self.position;
206            self.advance();
207            self.get(position)
208        }
209    }
210
211    /// Peek at the current token _without_ advancing the cursor position.
212    #[inline]
213    #[must_use]
214    pub fn peek(&self) -> Option<&Token> {
215        self.get(self.position)
216    }
217
218    /// Peek `ahead` number of tokens ahead of the current cursor position, _without_ advancing the
219    /// cursor position.
220    #[inline]
221    #[must_use]
222    pub fn peek_at(&self, ahead: u32) -> Option<&Token> {
223        self.get(self.position.saturating_add(ahead))
224    }
225}

258/// Describes a type of reference.
259/// Contains the result of parsing `&` or `&mut` from the [`TokenStream`].
260#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash)]
261pub enum RefType {
262    /// No specified reference.
263    None,
264    /// An immutable reference.
265    Ref,
266    /// A mutable reference.
267    RefMut,
268}

270impl RefType {
271    /// Return the [`Mutability`] of the [`RefType`].
272    /// I.e. `RefMut` = `Mutability::Mutable`, `Ref` & `None` = `Mutability::Immutable`.
273    #[inline]
274    #[must_use]
275    pub const fn mutability(self) -> Mutability {
276        match self {
277            Self::RefMut => Mutability::Mutable,
278            _ => Mutability::Immutable,
279        }
280    }
281
282    /// Returns `true` if `self` is a `reference`.
283    #[inline]
284    #[must_use]
285    pub const fn is_ref(self) -> bool {
286        matches!(self, Self::Ref | Self::RefMut)
287    }

290impl Parser<'_, '_> {
291    /// Returns true if the next 2 tokens from the cursor position are [`Punctuation::Colon`].
292    #[inline]
293    #[must_use]
294    fn is_double_colon(&self) -> bool {
295        self.check_kind_at(0, Punctuation::Colon) && self.check_kind_at(1, Punctuation::Colon)
296    }
297
298    /// Parses a double colon `"::"` or the "path" separator from the cursor position.
299    ///
300    /// This function advances the cursor position by the length of the path separator (2) if the
301    /// double colon is found.
302    /// # Returns
303    /// *   `Ok(true)` if a double colon was found and the cursor was advanced.
304    /// *   `Ok(false)` if a double colon wasn't found, but the token at the current cursor position
305    ///     _was not_ a single colon `':'`, and the cursor was not advanced.
306    /// *   `Err(ParseErrorKind::InvalidPath)` if the double colon was not found, but a single
307    ///     colon was, indicating a malformed path.
308    fn parse_double_colon(&mut self, path_start: u32) -> PResult<bool> {
309        if self.is_double_colon() {
310            self.cursor.advance_by(2);
311            Ok(true)
312        } else {
313            if self.check_kind(Punctuation::Colon) {
314                return Err(ParseError::new(
315                    ParseErrorKind::InvalidPath,
316                    path_start..self.cursor.get_current_source_end(),
317                ));
318            }
319            Ok(false)
320        }
321    }
322
323    pub fn parse_path_segment(&mut self) -> PResult<String> {
324        let token = self.peek()?.clone();
325        match token.kind {
326            TokenKind::Ident(ref ident_value) => {
327                self.cursor.advance();
328                Ok(ident_value.clone())
329            }
330            TokenKind::Keyword(Keyword::ThisTy) => {
331                self.cursor.advance();
332                Ok("Self".to_string())
333            }
334            k => Err(ParseError::new(
335                ParseErrorKind::ExpectedIdentifier(k),
336                self.cursor.current_span(),
337            )),
338        }
339    }
340
341    /// Parse a 'path' (I.e. `std::math::powf`) from the current cursor position,
342    /// into a path segment representation.
343    /// # Notes
344    /// Always expects atleast one `identifier`, can optionally start with '::', to force the path
345    /// to reference absolutely from the 'base'.
346    pub fn parse_path(&mut self) -> PResult<IdentPath> {
347        let path_start = self.cursor.get_current_source_start();
348        let mut path_segments = IdentPathSegments::new();
349        let root_relative = self.parse_double_colon(path_start)?;
350
351        path_segments.push(self.parse_path_segment()?);
352
353        while !self.cursor.is_end() {
354            if !self.parse_double_colon(path_start)? {
355                break;
356            }
357
358            // Testing
359            if self.check_kind(Punctuation::OpenBrace) {
360                break;
361            }
362
363            path_segments.push(self.parse_path_segment()?);
364        }
365
366        Ok(IdentPath::from_segments(path_segments, root_relative))
367    }

369    /// Parse a 'path' (I.e. `std::math::powf`) from the current cursor position,
370    /// into a path segment representation, also returns the span of the parsed path.
371    /// # Notes
372    /// Always expects atleast one `identifier`, can optionally start with '::', to force the path
373    /// to reference absolutely from the 'base'.
374    pub fn parse_spanned_path(&mut self) -> PResult<SpannedIdentPath> {
375        let path_span_start = self.begin_span();
376        let path = self.parse_path()?;
377        Ok(SpannedIdentPath {
378            path,
379            span: self.finish_span(path_span_start),
380        })
381    }

383    /// Parses a `&` and `&mut`
384    pub fn parse_ref_and_refmut(&mut self) -> PResult<RefType> {
385        if self.check_kind_advance_err(Punctuation::Ampersand)? {
386            if self.check_kind_advance_err(Keyword::Mut)? {
387                Ok(RefType::RefMut)
388            } else {
389                Ok(RefType::Ref)
390            }
391        } else {
392            // TODO: Check if next token is 'mut', if so, return an error.
393            Ok(RefType::None)
394        }
395    }

397    /// Parse an array or slice type from the current cursor position.
398    /// # Example
399    /// ```ignore
400    /// [Ty; N]
401    /// [Ty]
402    /// ```
403    /// # Notes
404    /// Must be verified the type being parsed is _NOT_ a unit type `()` by the caller.
405    pub fn parse_array_or_slice_ty(&mut self) -> PResult<Ty> {
406        let span_start = self.begin_span();
407        self.expect_kind(Punctuation::OpenBracket)?;
408
409        let element_type = Box::new(self.parse_ty()?);
410
411        #[allow(clippy::cast_sign_loss, clippy::cast_possible_truncation)]
412        if self.check_kind_advance(Punctuation::SemiColon) {
413            // Array..
414            match self.expect_literal()? {
415                LiteralKind::Integer(len) => {
416                    self.expect_kind(Punctuation::CloseBracket)?;
417                    Ok(Ty::Array(
418                        element_type,
419                        len as usize,
420                        self.finish_span(span_start),
421                    ))
422                }
423                LiteralKind::Float(_) => Err(ParseError::new(
424                    ParseErrorKind::InvalidArraySizeSpecifier,
425                    self.cursor.current_span(),
426                )),
427            }
428        } else {
429            // Slice..
430            self.expect_kind(Punctuation::CloseBracket)?;
431            Ok(Ty::Slice(element_type, self.finish_span(span_start)))
432        }
433    }

435    /// Parse a tuple type from the current cursor position.
436    /// # Example
437    /// ```ignore
438    /// (Ty, Ty, ...)
439    /// ```
440    /// # Notes
441    /// Must be verified the type being parsed is _NOT_ a unit type `()` by the caller.
442    pub fn parse_tuple_ty(&mut self) -> PResult<Ty> {
443        let span_start = self.begin_span();
444        self.expect_kind(Punctuation::OpenParen)?;
445
446        let mut tuple_types = vec![Box::new(self.parse_ty()?)];
447
448        while self.check_kind_advance(Punctuation::Comma) {
449            if self.check_kind(Punctuation::CloseParen) {
450                break;
451            }
452            tuple_types.push(Box::new(self.parse_ty()?));
453        }
454
455        self.expect_kind(Punctuation::CloseParen)?;
456        Ok(Ty::Tuple(tuple_types, self.finish_span(span_start)))
457    }

459    /// Parse a [`Ty`] from the current cursor position.
460    /// # Notes
461    /// Always expects atleast one `Identifier`.
462    pub fn parse_ty(&mut self) -> PResult<Ty> {
463        let span_start = self.begin_span();
464        let ref_type = self.parse_ref_and_refmut()?;
465
466        let root_type = match self.peek()?.kind {
467            TokenKind::Punctuation(Punctuation::OpenParen) => {
468                if self.check_kind_at(1, Punctuation::CloseParen) {
469                    let unit_start_span = self.begin_span();
470                    self.cursor.advance_by(2);
471                    Ty::Unit(self.finish_span(unit_start_span))
472                } else {
473                    self.parse_tuple_ty()?
474                }
475            }
476            TokenKind::Punctuation(Punctuation::OpenBracket) => self.parse_array_or_slice_ty()?,
477            TokenKind::Keyword(Keyword::ThisTy) => {
478                let self_span_start = self.begin_span();
479                self.cursor.advance();
480                Ty::This(self.finish_span(self_span_start))
481            }
482            _ => Ty::new(self.parse_spanned_path()?),
483        };
484
485        if ref_type.is_ref() {
486            Ok(Ty::Ref(
487                Box::new(root_type),
488                ref_type.mutability(),
489                self.finish_span(span_start),
490            ))
491        } else {
492            Ok(root_type)
493        }
494    }

496    /// Parse the type identifier for an `impl` block.
497    pub fn parse_impl_ty_ident(&mut self) -> PResult<SpannedIdentPath> {
498        // Case for unit type "()".
499        if self.check_kind(Punctuation::OpenParen) && self.check_kind_at(1, Punctuation::CloseParen)
500        {
501            let unit_span_start = self.begin_span();
502            self.cursor.advance_by(2);
503            let unit_path = IdentPath::from_segments_slice(&["()".to_string()], false);
504
505            return Ok(SpannedIdentPath::new(
506                unit_path,
507                self.finish_span(unit_span_start),
508            ));
509        }
510
511        self.parse_spanned_path()
512    }

514    /// Checks if the current keyword is `pub`, if it is, consume it and return `Ok(Visibility::Public)`,
515    /// otherwise `Ok(Visibility::Private)`, If there are no tokens, return `Err`.
516    pub fn parse_visibility(&mut self) -> PResult<Visibility> {
517        Ok(Visibility::from_is_public(
518            self.check_kind_advance_err(Keyword::Pub)?,
519        ))
520    }

522    /// Checks if the current keyword is `mut`, if it is, consume it and return `Ok(Mutability::Mutable)`,
523    /// otherwise `Ok(Mutability::Immutable)`, If there are no tokens, return `Err`.
524    pub fn parse_mutability(&mut self) -> PResult<Mutability> {
525        let mutable = matches!(self.peek()?.kind, TokenKind::Keyword(Keyword::Mut));
526        if mutable {
527            self.cursor.advance();
528        }
529        Ok(Mutability::from_is_mutable(mutable))
530    }

532    /// Parse the 'root' AST, starting from the beginning of the file.
533    /// This is the function you want to call to fully parse a token stream.
534    pub fn parse_root(&mut self) -> PResult<ASTRoot> {
535        let mut root_ast = ASTRoot::new_with_span(self.cursor.full_span().into());
536
537        while !self.cursor.is_end() {
538            let Some((_offset, token)) = self.find_next_significant_token()? else {
539                break;
540            };
541
542            if matches!(token.kind, TokenKind::Eof) {
543                self.cursor.advance();
544            } else if token.kind.can_start_item() {
545                root_ast.push_item(self.parse_item()?);
546            } else {
547                return Err(ParseError::new(
548                    ParseErrorKind::ExpectedItem(token.kind.clone()),
549                    token,
550                ));
551            }
552        }
553
554        Ok(root_ast)
555    }

560    /// Peek at the current token, return `true` if the [`TokenKind`] of the [`Token`] matches the
561    /// `expected_kind`
562    #[inline]
563    fn check_kind<T: Into<TokenKind>>(&self, expected_kind: T) -> bool {
564        let expected = expected_kind.into();
565        self.cursor.peek().is_some_and(|t| t.kind == expected)
566    }
567
568    /// Peek at the specified `offset`, return `true` if the [`TokenKind`] of the [`Token`] matches the
569    /// `expected_kind`
570    #[inline]
571    #[allow(dead_code)]
572    fn check_kind_at_err<T: Into<TokenKind>>(
573        &self,
574        offset: u32,
575        expected_kind: T,
576    ) -> PResult<bool> {
577        let expected = expected_kind.into();
578        Ok(self.peek_at(offset)?.kind == expected)
579    }

581    /// Peek at the current token, return `true` if the [`TokenKind`] of the [`Token`] matches the
582    /// `expected_kind`
583    #[inline]
584    fn check_kind_err<T: Into<TokenKind>>(&self, expected_kind: T) -> PResult<bool> {
585        let expected = expected_kind.into();
586        Ok(self.peek()?.kind == expected)
587    }

589    /// Peek at the specified `offset`, return `true` if the [`TokenKind`] of the [`Token`] matches the
590    /// `expected_kind`
591    #[inline]
592    fn check_kind_at<T: Into<TokenKind>>(&self, offset: u32, expected_kind: T) -> bool {
593        let expected = expected_kind.into();
594        self.cursor
595            .peek_at(offset)
596            .is_some_and(|t| t.kind == expected)
597    }

599    /// Peek at the current token, return `true` and advance the cursor if the [`TokenKind`] of the [`Token`]
600    /// matches the `expected_kind`
601    #[inline]
602    fn check_kind_advance<T: Into<TokenKind>>(&mut self, expected_kind: T) -> bool {
603        if self.check_kind(expected_kind) {
604            self.cursor.advance();
605            true
606        } else {
607            false
608        }
609    }

611    /// Peek at the current token, return `true` and advance the cursor if the [`TokenKind`] of the [`Token`]
612    /// matches the `expected_kind`
613    #[inline]
614    fn check_kind_advance_err<T: Into<TokenKind>>(&mut self, expected_kind: T) -> PResult<bool> {
615        if self.check_kind_err(expected_kind)? {
616            self.cursor.advance();
617            Ok(true)
618        } else {
619            Ok(false)
620        }
621    }

623    /// Peek at the current token, return `true` if the [`TokenKind`] of the [`Token`]
624    /// is an `Identifier`.
625    #[inline]
626    fn check_ident(&self) -> bool {
627        self.cursor
628            .peek()
629            .is_some_and(|t| matches!(t.kind, TokenKind::Ident(_)))
630    }

632    /// Peek at an offset from the current token, return `true` if the [`TokenKind`] of the [`Token`]
633    /// is an `Identifier`.
634    #[inline]
635    fn check_ident_at(&self, offset: u32) -> bool {
636        self.cursor
637            .peek_at(offset)
638            .is_some_and(|t| matches!(t.kind, TokenKind::Ident(_)))
639    }

641    /// Peek at the current token, check if the [`TokenKind`] of the [`Token`]
642    /// is an `Identifier`.
643    /// # Returns
644    /// `Some(ident_value)` and advance the cursor if it is an `Identifier`,
645    /// else `None`.
646    #[inline]
647    fn check_ident_advance(&mut self) -> Option<String> {
648        if self.check_ident() {
649            self.cursor.consume().and_then(|t| {
650                if let TokenKind::Ident(ident_value) = &t.kind {
651                    Some(ident_value.clone())
652                } else {
653                    None
654                }
655            })
656        } else {
657            None
658        }
659    }

661    /// Get a sequence of [`Punctuation`] tokens of length `N` from a specified offset from the current cursor
662    /// position.
663    /// # Returns
664    /// *   `Some(arr)` if the next `N` tokens from the specified offset are all [`Punctuation`] token kinds.
665    /// *   `None` if any of the next `N` tokens from the specified offset are _not_ of kind [`Punctuation`].
666    #[allow(clippy::cast_possible_truncation)]
667    #[inline]
668    fn get_punctuation_sequence<const N: usize>(&self, offset: u32) -> Option<[Punctuation; N]> {
669        let mut result = [Punctuation::Tilde; N];
670        for (i, out_p) in result.iter_mut().enumerate() {
671            let token = self.cursor.peek_at(offset + i as u32)?;
672
673            if let TokenKind::Punctuation(p) = token.kind {
674                *out_p = p;
675            } else {
676                return None;
677            }
678        }
679        Some(result)
680    }
681}

683// Expect..
684impl Parser<'_, '_> {
685    /// Expect that the token at the current cursor position is a literal.
686    ///
687    /// This function will advance the cursor by one position if a literal is found.
688    /// # Returns
689    /// *   `Ok(())` if the token at the cursor position is a literal.
690    /// *   `Err(ParseErrorKind::ExpectedLiteral)` if the token found at the current cursor position
691    ///     was not a literal.
692    /// *   `Err(ParseErrorKind::ExpectedTokenFoundNone)` if there are no more tokens in the
693    ///     stream, but it was expected that there were.
694    fn expect_literal(&mut self) -> PResult<LiteralKind> {
695        let token = self.peek()?.clone();
696        match &token.kind {
697            TokenKind::Literal(lit, ..) => {
698                self.cursor.advance();
699                Ok(*lit)
700            }
701            k => Err(ParseErrorKind::ExpectedLiteral(k.clone()).with_span(token)),
702        }
703    }
704
705    /// Expect that the token at the current cursor position is of a specified [`TokenKind`].
706    ///
707    /// This function will advance the cursor by one position if the `expected_kind` is found.
708    /// # Returns
709    /// *   `Ok(())` if the token at the cursor position is of the `expected_kind`.
710    /// *   `Err(ParseErrorKind::ExpectedToken)` if the token found at the current cursor position
711    ///     does not match the `expected_kind`.
712    /// *   `Err(ParseErrorKind::ExpectedTokenFoundNone)` if there are no more tokens in the
713    ///     stream, but it was expected that there were.
714    #[inline]
715    fn expect_kind<T: Into<TokenKind>>(&mut self, expected_kind: T) -> PResult<()> {
716        let expected = expected_kind.into();
717        if self.check_kind_advance(expected.clone()) {
718            Ok(())
719        } else if let Some(token) = self.cursor.peek() {
720            Err(ParseError::new(
721                ParseErrorKind::ExpectedToken(expected, token.kind.clone()),
722                token,
723            ))
724        } else {
725            Err(ParseError::new(
726                ParseErrorKind::ExpectedTokenFoundNone(expected),
727                self.cursor.eof_span(),
728            ))
729        }
730    }
731
732    /// Expect that the token at the current cursor position is an `Identifier`.
733    ///
734    /// This function will advance the cursor by one position if an `Identifier` is found.
735    /// # Returns
736    /// *   `Ok(ident_str)` if the token at the cursor position is an `Identifier`.
737    /// *   `Err(ParseErrorKind::ExpectedIdentifier)` if the token found at the current cursor position
738    ///     was not an `Identifier`.
739    /// *   `Err(ParseErrorKind::ExpectedIdentifierFoundNone)` if there are no more tokens in the
740    ///     stream, but it was expected that there were.
741    #[inline]
742    fn expect_ident(&mut self) -> PResult<String> {
743        self.check_ident_advance().ok_or_else(|| {
744            if let Some(token) = self.cursor.peek() {
745                ParseError::new(
746                    ParseErrorKind::ExpectedIdentifier(token.kind.clone()),
747                    token,
748                )
749            } else {
750                ParseError::new(
751                    ParseErrorKind::ExpectedIdentifierFoundNone,
752                    self.cursor.eof_span(),
753                )
754            }
755        })
756    }
757
758    /// Expect that the token at the current cursor position is an `Identifier`.
759    ///
760    /// This function will advance the cursor by one position if an `Identifier` is found.
761    /// # Returns
762    /// *   `Ok((ident_str, span))` if the token at the cursor position is an `Identifier`.
763    /// *   `Err(ParseErrorKind::ExpectedIdentifier)` if the token found at the current cursor position
764    ///     was not an `Identifier`.
765    /// *   `Err(ParseErrorKind::ExpectedIdentifierFoundNone)` if there are no more tokens in the
766    ///     stream, but it was expected that there were.
767    #[inline]
768    fn expect_ident_spanned(&mut self) -> PResult<SpannedIdent> {
769        let span_start = self.begin_span();
770        let ident = self.expect_ident()?;
771        Ok(SpannedIdent::new(ident, self.finish_span(span_start)))
772    }
773}

795// Implementation details..
796impl Parser<'_, '_> {
797    /// Checks if the token is valid, returning an `Err` if it isn't.
798    /// # Returns
799    /// *   `Ok(())` if the token is valid,
800    /// *   `Err(ParseError)` if the token is invalid.
801    fn check_token_invalid(token: &Token) -> PResult<()> {
802        match &token.kind {
803            TokenKind::InvalidIdent(ident) => Err(ParseError::new(
804                ParseErrorKind::InvalidIdentifier(ident.clone()),
805                token,
806            )),
807            TokenKind::InvalidLiteral(source) => Err(ParseError::new(
808                ParseErrorKind::InvalidLiteral(source.clone()),
809                token,
810            )),
811            TokenKind::Unknown => Err(ParseError::new(ParseErrorKind::UnknownToken, token)),
812            _ => Ok(()),
813        }
814    }
815
816    /// Find the next significant token, I.e. A token that isn't `mut` or something similar, and
817    /// that may identify which variant we should try to parse.
818    /// # Returns
819    /// *   `Ok(Some(offset_from_current_token, significant_token))` if there is no error found while
820    ///     checking tokens.
821    /// *   `Ok(None)` if no significant token was found, but there were no errors either.
822    /// *   `Err(ParseError)` if an invalid token was found while checking next tokens.
823    fn find_next_significant_token(&self) -> PResult<Option<(u32, &Token)>> {
824        self.find_next_significant_token_with_offset(0)
825    }
826
827    /// Find the next significant token, I.e. A token that isn't `mut` or something similar, and
828    /// that may identify which variant we should try to parse, starting from a beginning offset
829    /// from the current cursor position `offset`.
830    /// # Returns
831    /// *   `Ok(Some(offset_from_current_token, significant_token))` if there is no error found while
832    ///     checking tokens.
833    /// *   `Ok(None)` if no significant token was found, but there were no errors either.
834    /// *   `Err(ParseError)` if an invalid token was found while checking next tokens.
835    fn find_next_significant_token_with_offset(
836        &self,
837        offset: u32,
838    ) -> PResult<Option<(u32, &Token)>> {
839        let remaining = self.cursor.remaining().saturating_sub(offset);
840        for o in 0..remaining {
841            if let Some(t) = self.cursor.peek_at(offset + o) {
842                Self::check_token_invalid(t)?;
843                if t.kind.is_significant() {
844                    return Ok(Some((offset + o, t)));
845                }
846            } else {
847                break;
848            }
849        }
850        Ok(None)
851    }
852
853    /// Find the next significant token, I.e. A token that isn't `mut` or something similar, and
854    /// that may identify which variant we should try to parse, starting from a beginning offset
855    /// from the current cursor position `offset`.
856    /// # Returns
857    /// *   `Ok(offset_from_current_token, significant_token)` if there is no error found while
858    ///     checking tokens.
859    /// *   `Err(ParseError)` if an invalid token was found while checking next tokens, or if there
860    ///     were no more tokens available.
861    #[inline]
862    fn expect_next_significant_token_with_offset(&self, offset: u32) -> PResult<(u32, &Token)> {
863        self.find_next_significant_token_with_offset(offset)?
864            .ok_or_else(|| self.no_token_error())
865    }
866
867    /// Find the next significant token, I.e. A token that isn't `mut` or something similar, and
868    /// that may identify which variant we should try to parse.
869    /// # Returns
870    /// *   `Ok(offset_from_current_token, significant_token)` if there is no error found while
871    ///     checking tokens.
872    /// *   `Err(ParseError)` if an invalid token was found while checking next tokens, or if there
873    ///     were no more tokens available.
874    #[inline]
875    fn expect_next_significant_token(&self) -> PResult<(u32, &Token)> {
876        self.expect_next_significant_token_with_offset(0)
877    }
878}

880// Parsing 'block' like structures.
881impl Parser<'_, '_> {
882    /// Parses a 'block'-like pattern, where the expected token stream to parse resembles:
883    /// ```ignore
884    /// <OpenPunctuation>
885    ///     <Element>+
886    /// <ClosePunctuation>
887    /// ````
888    /// # Notes
889    /// This function expects that the `<OpenPunctuation>` has already been consumed and that the
890    /// cursor is at the start of the first element in the block.
891    ///
892    /// This function allows for empty blocks.
893    /// # Generic Parameters
894    /// `T` is the type of the element the block will parse.
895    /// `F` is a function that will actually do the parsing of each element, and the function is
896    /// passed `self`.
897    fn parse_block_like_no_delimiter<T, F: FnMut(&mut Self) -> PResult<T>>(
898        &mut self,
899        close: Punctuation,
900        mut f: F,
901    ) -> PResult<Vec<T>> {
902        let mut elements = Vec::new();
903
904        if !self.check_kind_advance(close) {
905            elements.push(f(self)?);
906
907            while !self.cursor.is_end() {
908                if self.check_kind_advance(close) {
909                    break;
910                }
911                elements.push(f(self)?);
912            }
913        }
914
915        Ok(elements)
916    }
917
918    /// Parses a 'block'-like pattern, where the expected token stream to parse resembles:
919    /// ```ignore
920    /// <OpenPunctuation>
921    ///     <Element><Delimiter>
922    ///     <Element><Delimiter>?
923    /// <ClosePunctuation>
924    /// ````
925    /// # Notes
926    /// This function expects that the `<OpenPunctuation>` has already been consumed and that the
927    /// cursor is at the start of the first element in the block.
928    ///
929    /// This function allows for empty blocks.
930    /// # Generic Parameters
931    /// If `TRAIL_DELIM` is set `true`, the algorithm will allow a trailing delimiter after the
932    /// last element but before the closing [`Punctuation`], otherwise this will cause an error.
933    /// `T` is the type of the element the block will parse.
934    /// `F` is a function that will actually do the parsing of each element, and the function is
935    /// passed `self`.
936    fn parse_block_like_impl<const TRAIL_DELIM: bool, T, F: FnMut(&mut Self) -> PResult<T>>(
937        &mut self,
938        delimiter: Punctuation,
939        close: Punctuation,
940        mut f: F,
941    ) -> PResult<Vec<T>> {
942        let mut elements = Vec::new();
943
944        if !self.check_kind_advance(close) {
945            elements.push(f(self)?);
946
947            while !self.cursor.is_end() {
948                if self.check_kind_advance(close) {
949                    break;
950                }
951                if TRAIL_DELIM && self.check_kind(delimiter) && self.check_kind_at(1, close) {
952                    self.cursor.advance_by(2);
953                    break;
954                }
955                self.expect_kind(delimiter)?;
956
957                elements.push(f(self)?);
958            }
959        }
960
961        Ok(elements)
962    }
963
964    /// Parses a 'block'-like pattern, where the expected token stream to parse resembles:
965    /// ```ignore
966    /// <OpenPunctuation>
967    ///     <Element><Delimiter>
968    ///     <Element><Delimiter>?
969    /// <ClosePunctuation>
970    /// ````
971    /// # Notes
972    /// This function expects that the `<OpenPunctuation>` has already been consumed and that the
973    /// cursor is at the start of the first element in the block.
974    ///
975    /// This function allows for empty blocks.
976    ///
977    /// This function allows for trailing delimiters on the last element before the end of the
978    /// `block`.
979    /// # Generic Parameters
980    /// `T` is the type of the element the block will parse.
981    /// `F` is a function that will actually do the parsing of each element, and the function is
982    /// passed `self`.
983    #[inline]
984    fn parse_block_like<T, F: FnMut(&mut Self) -> PResult<T>>(
985        &mut self,
986        delimiter: Punctuation,
987        close: Punctuation,
988        f: F,
989    ) -> PResult<Vec<T>> {
990        self.parse_block_like_impl::<true, T, F>(delimiter, close, f)
991    }
992
993    /// Parses a 'block'-like pattern, where the expected token stream to parse resembles:
994    /// ```ignore
995    /// <OpenPunctuation>
996    ///     <Element><Delimiter>
997    ///     <Element><Delimiter>?
998    /// <ClosePunctuation>
999    /// ````
1000    /// # Notes
1001    /// This function expects that the `<OpenPunctuation>` has already been consumed and that the
1002    /// cursor is at the start of the first element in the block.
1003    ///
1004    /// This function allows for empty blocks.
1005    ///
1006    /// This function does not allow for trailing delimiters on the last element before the end of the
1007    /// `block`.
1008    /// # Generic Parameters
1009    /// `T` is the type of the element the block will parse.
1010    /// `F` is a function that will actually do the parsing of each element, and the function is
1011    /// passed `self`.
1012    #[inline]
1013    #[allow(dead_code)]
1014    fn parse_block_like_no_trail<T, F: FnMut(&mut Self) -> PResult<T>>(
1015        &mut self,
1016        delimiter: Punctuation,
1017        close: Punctuation,
1018        f: F,
1019    ) -> PResult<Vec<T>> {
1020        self.parse_block_like_impl::<false, T, F>(delimiter, close, f)
1021    }
1022}

1055/// Parse AST from a [`Token`] iterator.
1056/// # Note
1057/// This function will collect the iterator into a [`Vec<Token>`].
1058/// # Errors
1059/// This function will return an error if any part of parsing the token stream fails.
1060/// The returned error will contain information about where in the source code the error occurred,
1061/// and a message about the nature of the error.
1062pub fn parse_from_tokens(tokens: impl Iterator<Item = Token>) -> PResult<ASTRoot> {
1063    // TODO: Probably try a method to not collect the iterator in the future, this is easier for
1064    // now though.
1065    let token_list = tokens.collect::<Vec<Token>>();
1066    let context = ParserContext {};
1067
1068    let mut parser = Parser::from_cursor(TokenCursor::new(&token_list), &context);
1069
1070    parser.parse_root()
1071}

1074/// # Note
1075/// This function will collect the iterator into a [`Vec<Token>`].
1076///
1077/// This function also strips all comments and documentation from the code.
1078/// # Errors

1080/// The returned error will contain information about where in the source code the error occurred,
