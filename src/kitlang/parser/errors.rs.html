9/// Parse error result, has [`ParseError`] as the error variant.

12#[derive(Error, Debug, Clone, PartialEq, PartialOrd)]
13pub enum ParseErrorKind {
14    #[error("Invalid string literal")]
15    InvalidStringLiteral,
16    #[error("Invalid identifier: {0}")]
17    InvalidIdentifier(String),
18    #[error("Invalid literal: {0}")]
19    InvalidLiteral(String),
20    #[error("Unknown token found")]
21    UnknownToken,
22
23    #[error("Wrong token type found: {0:?}")]
24    WrongTokenKind(TokenKind),
25    #[error("Expected token: {0:?}, found: {1:?}")]
26    ExpectedToken(TokenKind, TokenKind),
27    #[error("Expected token: {0:?}, but none was found")]
28    ExpectedTokenFoundNone(TokenKind),
29
30    #[error("Invalid expression atom: {0:?}")]
31    InvalidExpressionAtom(TokenKind),
32
33    #[error("Expected identifier, found: {0:?}")]
34    ExpectedIdentifier(TokenKind),
35    #[error("Expected identifier, but none was found")]
36    ExpectedIdentifierFoundNone,
37
38    #[error("Found a non-native global function, only native global functions are allowed")]
39    NonNativeGlobalFunction,
40
41    #[error("Parameter 'self' must be the first argument in the function parameters")]
42    SelfMustBeFirstArgument,
43    #[error(
44        "Parameter 'self' cannot be used in free functions, it must be used within an associated 'impl' block"
45    )]
46    SelfMustBeUsedInAMethod,
47
48    #[error("Invalid assignment operator: {0:?}")]
49    InvalidAssignmentOperator(BinaryOpKind),

51    #[error("Expected Semi-colon, expressions can only be at the end of a block")]
52    ExpectedSemiColon,
53
54    #[error("Expected an item, found: {0:?}")]
55    ExpectedItem(TokenKind),
56
57    #[error("Unexpected token: {0:?}")]
58    UnexpectedToken(TokenKind),
59
60    #[error("Invalid path, path separator must be '::' and all segments must be identifiers")]
61    InvalidPath,
62
63    #[error("Found an invalid binary operator")]
64    InvalidBinaryOperator,
65
66    #[error("Tried to parse AST type, but there were no available tokens")]
67    NoTokens,
68    #[error("Unterminated string literal.")]
69    UnterminatedStringLiteral,

74    #[error("Feature not implemented: {0}")]

80impl ParseErrorKind {
81    /// Create a new [`ParseError`] from this kind and the given [`SourceSpan`].
82    #[inline]
83    #[must_use]
84    pub fn with_span(&self, span: impl Into<SourceSpan>) -> ParseError {
85        ParseError::new(self.clone(), span.into())
86    }
87
88    /// Create a new [`ParseError`] from this kind and a null [`SourceSpan`].
89    #[inline]
90    #[must_use]
91    pub fn with_no_span(&self) -> ParseError {
92        ParseError::new(self.clone(), SourceSpan::null_span())
93    }
94}

96/// Represents an error while parsing the tokens produced by the lexer into an AST.
97#[derive(Debug, Error)]
98pub struct ParseError {
99    pub span: SourceSpan,
100    #[source]
101    pub error_kind: ParseErrorKind,
102}
103
104impl ParseError {
105    /// Create a new [`ParseError`] from the given kind and source span.
106    #[inline]
107    #[must_use]
108    pub fn new(kind: impl Into<ParseErrorKind>, span: impl Into<SourceSpan>) -> Self {
109        Self {
110            error_kind: kind.into(),
111            span: span.into(),
112        }
113    }
114
115    /// Create a new [`ParseError`] indicating that there were no tokens to parse.
116    #[inline]
117    #[must_use]
118    pub fn no_tokens(span: impl Into<SourceSpan>) -> Self {
119        Self::new(ParseErrorKind::NoTokens, span)
120    }
121
122    /// Format the error as an error message, given the source code string.
123    #[inline]
124    #[must_use]
125    pub fn format_as_error_message(&self, source_string: &str) -> String {
126        SpannedErrorBuilder::new(source_string, self.span)
127            .generate_highlight()
128            .print_header_line(format!("Error: {}", self.error_kind))
129            .generate_output()
130    }
131}
132
133impl Display for ParseError {
134    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
135        write!(
136            f,
137            "Error: {}, Location: [{}..{}]",
138            self.error_kind, self.span.start, self.span.end
139        )
140    }
