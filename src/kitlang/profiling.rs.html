1//! Small utilities module for profiling and displaying the execution time of code blocks in an easily read format.
2
3use std::time::Duration;
4#[cfg(not(feature = "webasm"))]
5use std::time::Instant;
6#[cfg(feature = "webasm")]
7use web_time::Instant;
8
9use log::debug;
10
11use humanize_duration::{Truncate, prelude::*};
12
13/// Measures the execution time of a provided lambda function (closure) and returns the result of
14/// the provided lambda, along with the elapsed execution time.
15#[inline]
16#[must_use]
17pub fn measure_execution<F, R>(f: F) -> (R, Duration)
18where
19    F: FnOnce() -> R,
20{
21    let start = Instant::now();
22    let result = f();
23    let elapsed = start.elapsed();
24    (result, elapsed)
25}
26
27/// Converts a [`Duration`] into a human readable string.
28#[inline]
29#[must_use]
30pub fn format_duration(duration: Duration) -> String {
31    duration.human(Truncate::Nano).to_string()
32}
33
34/// Measures the execution time of a provided lambda function (closure) and returns the result of
35/// the provided lambda, prints the time elapsed while executing before returning the result.
36#[inline]
37#[must_use]
38pub fn print_execution_named<F, R>(name: &str, f: F) -> R
39where
40    F: FnOnce() -> R,
41{
42    let (result, duration) = measure_execution(f);
43    debug!(
44        "[Profiling] '{}' executed in {}.",
45        name,
46        format_duration(duration)
47    );
48    result
49}
