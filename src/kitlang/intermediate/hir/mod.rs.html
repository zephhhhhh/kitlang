1//! The HIR module provides the primary IR structure for the compiler.
2//! It transforms the AST into a flatter representation optimized for analysis passes like resolution and type checking.
3//!
4//! The [`HLIR`] organizes nodes into "owning nodes" (top-level items like functions, structs, impls) and their owned
5//! nodes (expressions, statements), each addressable via [`HirId`]. This structure enables efficient lookups and
6//! storage of side-channel data like type information without deep tree traversals, and having to store a way of navigating
7//! this tree structure.
8//!
9//! This module defines core ID types ([`HirId`], [`OwnerDefId`], [`LocalDefId`]), the [`HLIR`] container,
10//! program metadata ([`ProgramMetaData`]), and utilities for lowering, [traversing](crate::intermediate::hir::visitor),
11//! and manipulating the HIR.
12

35/// Definition ID that is only valid when relative to the "owner".
36#[derive(Default, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash)]
37pub struct LocalDefId(pub u32);
38
39impl LocalDefId {
40    /// Placeholder ID constant.
41    pub const PLACEHOLDER_ID: Self = Self(u32::MAX);
42
43    /// Check if this is a placeholder ID. I.e. (`self == u32::MAX`).
44    #[inline]
45    #[must_use]
46    pub fn is_placeholder(self) -> bool {
47        self == Self::PLACEHOLDER_ID
48    }
49}

53        write!(f, "LocalDefId({})", self.0)

81/// Definition ID that is relative to a specified module.
82#[derive(Default, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash)]
83pub struct DefId {
84    pub module_id: u32,
85    pub def_id: LocalDefId,
86}
87
88impl DefId {
89    // Placeholder ID constant.
90    pub const PLACEHOLDER_ID: Self = Self {
91        module_id: u32::MAX,
92        def_id: LocalDefId::PLACEHOLDER_ID,
93    };
94
95    /// Check if this is a placeholder ID. I.e. `module_id == u32::MAX` or `def_id == u32::MAX`.
96    #[inline]
97    #[must_use]
98    pub fn is_placeholder(self) -> bool {
99        self.module_id == u32::MAX || self.def_id.is_placeholder()
100    }
101}

104    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
105        write!(f, "DefId({} : {:?})", self.module_id, self.def_id)
106    }

109/// Definition ID that describes which ID into the modules `"path_table"` owns the node, and which
110/// index into the owners body the node refers to.
111#[derive(Default, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash)]
112pub struct HirId {
113    /// Index into the HLIR owning node list of the owner of this node.
114    pub owner: OwnerDefId,
115    /// Index into the owning node's HIR node list.
116    pub id: LocalDefId,
117}

143impl HirId {
144    /// Placeholder ID constant.
145    pub const PLACEHOLDER_ID: Self = Self {
146        owner: OwnerDefId::PLACEHOLDER_ID,
147        id: LocalDefId::PLACEHOLDER_ID,
148    };
149
150    /// Check if this is a placeholder ID. I.e. `owner == u32::MAX` or `id == u32::MAX`.
151    #[inline]

153    pub fn is_placeholder(self) -> bool {
154        self.owner.is_placeholder() || self.id.is_placeholder()
155    }

158impl HirId {
159    /// Get the next ID on the current owner by incrementing the local ID by 1.
160    #[inline]
161    #[must_use]
162    pub const fn next_id(mut self) -> Self {
163        self.id = LocalDefId(self.id.0 + 1);
164        self
165    }

171    }

174/// High Level Intermediate Representation of the program.
175///
176/// Contains all HIR nodes in owning nodes, which are organized in a structure where
177/// top level items such as structures, impls, etc each have their own owning node.
178///
179/// And all lower level nodes such as statements, expressions, etc are stored within those owning nodes.
180///
181/// In this fashion, the HIR can represent the entire program in a structure that is much flatter
182/// than the tree structure of the AST, while still maintaining the hierarchical relationships.
183///
184/// This means that all nodes have a small `HirId` that can be used to reference them, without needing
185/// to traverse a large tree structure.
186///
187/// This makes lookups and storing side channel data such as type information much easier and more efficient.
188#[derive(Default, Clone)]
189pub struct HLIR {
190    /// List of all owning nodes in the HIR.
191    pub owner_nodes: Vec<OwningNode>,
192}
193
194impl HLIR {
195    /// Get the total count of owner nodes in the HIR.
196    #[allow(clippy::cast_possible_truncation)]
197    #[inline]
198    #[must_use]
199    pub const fn owner_node_count(&self) -> u32 {
200        self.owner_nodes.len() as u32
201    }
202
203    /// Get an iterator over all owner IDs in the HIR.
204    #[inline]
205    pub fn owner_id_iter(&self) -> impl Iterator<Item = OwnerDefId> {
206        (0..self.owner_node_count()).map(OwnerDefId)
207    }
208
209    /// Get a reference to the root module owning node.
210    #[inline]
211    #[must_use]
212    pub fn root_module(&self) -> Option<&OwningNode> {
213        self.owning_node(OwnerDefId::ROOT_NODE)
214    }
215
216    /// Get a mutable reference to the root module owning node.
217    #[inline]
218    #[must_use]
219    pub fn root_module_mut(&mut self) -> Option<&mut OwningNode> {
220        self.owning_node_mut(OwnerDefId::ROOT_NODE)
221    }
222
223    /// Get a reference to the owning node with a specified [`OwnerDefId`].
224    #[inline]
225    #[must_use]
226    pub fn owning_node(&self, id: OwnerDefId) -> Option<&OwningNode> {
227        self.owner_nodes.get(id.0 as usize)
228    }
229
230    /// Get a mutable reference to the owning node with a specified [`OwnerDefId`].
231    #[inline]
232    #[must_use]
233    pub fn owning_node_mut(&mut self, id: OwnerDefId) -> Option<&mut OwningNode> {
234        self.owner_nodes.get_mut(id.0 as usize)
235    }
236
237    /// Get a reference to the owning node without checking for existence.
238    /// # Panics
239    /// Panics if the node does not exist.
240    #[inline]
241    #[must_use]
242    pub fn owning_node_unchecked(&self, id: OwnerDefId) -> &OwningNode {
243        self.owner_nodes.get(id.0 as usize).expect("Node exists.")
244    }
245
246    /// Get a mutable reference to the owning node without checking for existence.
247    /// # Panics
248    /// Panics if the node does not exist.
249    #[inline]
250    #[must_use]
251    pub fn owning_node_mut_unchecked(&mut self, id: OwnerDefId) -> &mut OwningNode {
252        self.owner_nodes
253            .get_mut(id.0 as usize)
254            .expect("Node exists.")
255    }
256
257    /// Get the next available owner ID for inserting a new owning node.
258    #[inline]
259    #[must_use]
260    pub const fn next_owner_id(&self) -> OwnerDefId {
261        OwnerDefId(self.owner_node_count())
262    }
263
264    /// Insert a new owning node into the HIR, returning its assigned [`OwnerDefId`].
265    #[inline]
266    pub fn insert_owning_node(&mut self, mut owner_node: OwningNode) -> OwnerDefId {
267        let next_id = self.next_owner_id();
268        owner_node.set_owner_id(next_id);
269        self.owner_nodes.push(owner_node);
270        next_id
271    }
272
273    /// Insert a new owning node into the HIR, updating its parent's item list,
274    /// and returning its assigned [`OwnerDefId`].
275    #[inline]
276    pub fn insert_owning_node_with_parent(
277        &mut self,
278        owner_node: OwningNode,
279        parent_id: OwnerDefId,
280    ) -> OwnerDefId {
281        let is_normal_item = matches!(&owner_node.kind, OwningNodeKind::Item(_));
282        let new_node_id = self.insert_owning_node(owner_node);
283        if is_normal_item {
284            self.update_parent_item_list(new_node_id, parent_id);
285        }
286        new_node_id
287    }
288
289    /// Push a new item to the parent owning node's item list.
290    fn update_parent_item_list(&mut self, new_node: OwnerDefId, parent_id: OwnerDefId) {
291        if let Some(parent_module) = self
292            .owning_node_mut(parent_id)
293            .and_then(|o| o.hir_module_mut())
294        {
295            parent_module.item_ids.push(new_node);
296        }
297    }
298}

300impl HLIR {
301    /// Get a reference to the [`HirNode`] with a specified [`HirId`].
302    #[inline]
303    #[must_use]

305        self.owning_node(hir_id.owner)?.get_hir_node(hir_id.id)

308    /// Get a reference to the [`HirNode`] with a specified [`HirId`] without checking for existence.
309    /// # Panics
310    /// Panics if the node does not exist.
311    #[inline]
312    #[must_use]
313    pub fn get_hir_node_unchecked(&self, hir_id: HirId) -> &HirNode {
314        self.owning_node_unchecked(hir_id.owner)
315            .get_hir_node(hir_id.id)
316            .expect("HIRNode exists.")
317    }
318
319    /// Get a mutable reference to the [`HirNode`] with a specified [`HirId`].
320    #[inline]
321    #[must_use]
322    pub fn get_hir_node_mut(&mut self, hir_id: HirId) -> Option<&mut HirNode> {
323        self.owning_node_mut(hir_id.owner)?
324            .get_hir_node_mut(hir_id.id)
325    }
326
327    /// Get a mutable reference to the [`HirNode`] with a specified [`HirId`] without checking for existence.
328    /// # Panics
329    /// Panics if the node does not exist.
330    #[inline]
331    #[must_use]
332    pub fn get_hir_node_mut_unchecked(&mut self, hir_id: HirId) -> &mut HirNode {
333        self.owning_node_mut_unchecked(hir_id.owner)
334            .get_hir_node_mut(hir_id.id)
335            .expect("HIRNode exists.")
336    }
337
338    /// Get the next available [`HirId`] for a specified owner.
339    #[inline]
340    #[must_use]
341    pub fn next_hir_id_on(&self, owner_id: impl Into<OwnerDefId>) -> HirId {
342        self.owning_node(owner_id.into())
343            .map_or(HirId::PLACEHOLDER_ID, nodes::OwningNode::next_hir_id)
344    }
345
346    /// Insert a new [`HirNode`] into the HIR under the specified owner, returning its assigned [`HirId`].
347    #[inline]
348    pub fn insert_hir_node(
349        &mut self,
350        owner_id: impl Into<OwnerDefId>,
351        hir_node: HirNode,
352    ) -> Option<HirId> {
353        let owner = owner_id.into();
354        let local_id = self.owning_node_mut(owner)?.insert_hir_node(hir_node);
355        Some(HirId {
356            owner,
357            id: local_id,
358        })
359    }

423/// Contains metadata about the program stored in HIR, such as type information, namespace etc.
424#[derive(Debug, Default, Clone)]
425pub struct ProgramMetaData {
426    /// The program's namespace information, I.e. all defined modules, types, functions, etc.
427    pub namespace: RootNamespace,
428    /// The program's type registry, containing all defined types.
429    pub type_registry: TypeRegistry,
430    /// The program's type map, containing side channel
431    /// information about what types a given [`HirId`] evaluates to.
432    pub type_map: TypeMap,
433}
434
435impl ProgramMetaData {
436    /// Get a reference to the [`Namespace`] defined at a given `path`, if it exists.
437    #[inline]
438    #[must_use]
439    pub fn get_namespace(&self, path: &IdentPath) -> Option<&Namespace> {
440        self.namespace.find_definition(path)
441    }
442
443    /// Get a mutable reference to the [`Namespace`] defined at a given `path`, if it exists.
444    #[inline]
445    #[must_use]
446    pub fn get_namespace_mut(&mut self, path: &IdentPath) -> Option<&mut Namespace> {
447        self.namespace.find_definition_mut(path)
448    }
449
450    /// Find a method in the namespace given its defining path, data type identifier, and method identifier.
451    /// # Returns
452    /// The method namespace if found, otherwise `None`.
453    #[inline]
454    #[must_use]
455    pub fn find_method(
456        &self,
457        defined_in: &IdentPath,
458        data_type_ident: &str,
459        method_ident: &str,
460    ) -> Option<&Namespace> {
461        self.namespace
462            .find_method(defined_in, data_type_ident, method_ident)
463    }
464
465    /// Find a method owner definition ID in the namespace given its defining path, data type identifier, and method identifier.
466    /// # Returns
467    /// The method owner definition ID if found, otherwise `None`.
468    #[inline]
469    #[must_use]
470    pub fn find_method_owner_def(
471        &self,
472        defined_in: &IdentPath,
473        data_type_ident: &str,
474        method_ident: &str,
475    ) -> Option<OwnerDefId> {
476        self.namespace
477            .find_method_owner_def(defined_in, data_type_ident, method_ident)
478    }
479
480    /// Find the owner definition ID for a method defined on an ADT.
481    /// # Returns
482    /// The method owner definition ID if found, otherwise `None`.
483    #[inline]
484    #[must_use]
485    pub fn find_adt_method_owner_def(
486        &self,
487        adt: &ADTTypeInfo,
488        method_ident: &str,
489    ) -> Option<OwnerDefId> {
490        self.find_method_owner_def(&adt.defined_in, adt.type_ident.str(), method_ident)
491    }
492
493    /// Find the owner definition ID for a method defined on a given type.
494    /// # Returns
495    /// The method owner definition ID if found, otherwise `None`.
496    #[inline]
497    #[must_use]
498    pub fn find_ty_method_owner_def(&self, ty: &Type, method_ident: &str) -> Option<OwnerDefId> {
499        let Type::Resolved(resolved_ty) = ty else {
500            return None;
501        };
502        match resolved_ty {
503            KitTy::Abstract(adt_id) => {
504                let adt = self.type_registry.get_from_type_id(*adt_id)?;
505                self.find_method_owner_def(&adt.defined_in, adt.type_ident.str(), method_ident)
506            }
507            t => self.find_method_owner_def(
508                &IdentPath::new_empty(true),
509                &t.to_type_str()?,
510                method_ident,
511            ),
512        }
513    }
514
515    /// Try to get the type name of a given type.
516    /// # Returns
517    /// The type name if it can be determined, otherwise `None`.
518    #[inline]
519    #[must_use]
520    pub fn try_type_name(&self, ty: impl Into<Type>) -> Option<String> {
521        match ty.into() {
522            Type::Unresolved(ty) => Some(ty.get_type_ident()),
523            Type::Resolved(KitTy::Abstract(ty_id)) => {
524                let abs_ty = self.type_registry.get_from_type_id(ty_id)?;
525                let type_path = abs_ty.defined_in.extend_ident(&abs_ty.type_ident.ident);
526                Some(type_path.to_string())
527            }
528            Type::Resolved(kit_ty) => kit_ty.to_type_str(),
529        }
530    }
531
532    /// Get the type name of a given type, or `"??"` if it cannot be determined.
533    /// # Returns
534    /// The type name, or `"??"` if the type name cannot be determined.
535    #[inline]
536    #[must_use]
537    pub fn type_name(&self, ty: impl Into<Type>) -> String {
538        self.try_type_name(ty).unwrap_or_else(|| String::from("??"))
539    }
540}

552/// Lower the output of the parser stage to HIR.
553/// # Note
554/// This function does not do any later processing.
555/// # Errors
556/// This function will return an error if any part of the AST cannot be lowered properly.
557/// The returned error will contain a diagnostic message indicating the nature and location of any failures,
558/// as well as which stage of lowering it occurred in.
559pub fn lower_ast_to_hir(ast: &ASTRoot) -> LowerResult<HLIR> {
560    lowerer::lower_ast_to_hir(ast)
561}
562
563/// Lower the output of the parser state to HIR, using a provided metadata structure.
564/// # Note
565/// Unlike `lower_ast_to_hir`, this function does do all HIR processing.
566/// (Type checking, resolution, etc).
567/// # Errors
568/// This function will return an error if any part of the AST cannot be lowered properly.
569/// The returned error will contain a diagnostic message indicating the nature and location of any failures,
570/// as well as which stage of lowering it occurred in.
571pub fn parse_ast_to_hir_processed_with(
572    ast: &ASTRoot,
573    meta_data: &mut ProgramMetaData,
574) -> LowerResult<HLIR> {
575    let mut hlir = lower_ast_to_hir(ast)?;
576
577    // Resolution..
578    resolve_paths(&mut hlir, meta_data)
579        .map_err(|e| LoweringErrorKind::ResolverError(e).with_no_span())?;
580
581    // Type checking..
582    run_type_checker(&mut hlir, meta_data)?;
583
584    Ok(hlir)
585}

587/// Lower the output of the parser state to HIR.
588/// # Note
589/// Unlike `lower_ast_to_hir`, this function does do all HIR processing.
590/// (Type checking, resolution, etc).
591/// # Errors
592/// This function will return an error if any part of the AST cannot be lowered properly.
593/// The returned error will contain a diagnostic message indicating the nature and location of any failures,
594/// as well as which stage of lowering it occurred in.
595pub fn parse_ast_to_hir_processed(ast: &ASTRoot) -> LowerResult<(ProgramMetaData, HLIR)> {
596    let mut meta_data = ProgramMetaData::default();
597
598    let hlir = parse_ast_to_hir_processed_with(ast, &mut meta_data)?;
599
600    Ok((meta_data, hlir))
