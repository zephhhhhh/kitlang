1//! The purpose of this module is to resolve type references during the resolution phase of the compiler.
2//! A type reference is any place where a type is referenced by its path, such as in function parameters,
3//! return types, let bindings, or struct field definitions.
4//!
5//! This module walks the [`HLIR`] and converts unresolved type paths into resolved type IDs that can be
6//! looked up in the type registry. This includes resolving types within structs, functions, let statements,
7//! and impl blocks, as well as handling `this` type annotations within impl contexts.
8//!
9//! This module is exclusively focused on type reference resolution. Other forms of reference resolution,
10//! such as local variable resolution or module/definition resolution, are handled elsewhere.
11

30/// A resolver that walks the [`HLIR`] and resolves type references.
31/// This struct maintains state about the current path context and any errors encountered during resolution.
32///
33/// This stage runs after all ADTs have been registered in the [`TypeRegistry`], allowing it to resolve types to
34/// their corresponding [`TypeID`]s.
35///
36/// # Example
37/// ```ignore
38/// struct ExampleStruct { ... }
39///
40/// fn main() {
41///     let s = ExampleStruct { ... };
42/// }
43///
44/// fn example_function(param: ExampleStruct) -> ExampleStruct { ... }
45/// ```
46/// In the above example, the [`TypeResolver`] would resolve the type reference `ExampleStruct` in the `let` statement
47/// to the corresponding [`TypeID`] in the type registry.
48///
49/// It would also resolve the parameter and return types of `example_function` to the appropriate [`TypeID`].
50struct TypeResolver<'a> {
51    /// The current implementation type context, if within an `impl` block.
52    /// This is used to resolve `self` type annotations.
53    pub current_impl: Option<Type>,
54    /// The stack of identifier paths representing the current path of where we are in the [`HLIR`].
55    pub current_path: IdentPath,
56
57    pub meta: &'a mut ProgramMetaData,
58
59    /// A collection of resolver errors encountered during type resolution.
60    /// These errors are accumulated and reported after the resolution process.
61    /// If there are any errors in here after the process is complete, the resolution is considered to have failed.
62    pub errors: Vec<ResolverError>,
63}
64
65impl<'a> TypeResolver<'a> {
66    pub fn new(meta: &'a mut ProgramMetaData) -> Self {
67        Self {
68            current_impl: None,
69            current_path: IdentPath::ROOT,
70            meta,
71            errors: Vec::new(),
72        }
73    }
74}
75
76impl TypeResolver<'_> {
77    /// `Entrypoint` for resolving types in the given [`HLIR`].
78    /// This function performs a mutable walk over the [`HLIR`], resolving type references.
79    /// # Returns
80    /// * `Ok(())` if all type references were resolved successfully.
81    /// * `Err(ResolverError)` if there were any resolution errors.
82    /// # Errors
83    /// This function will return an error if any type references could not be resolved.
84    /// The returned error may contain multiple resolution errors if there were multiple failures.
85    pub fn resolve_types(&mut self, hlir: &mut HLIR) -> ResolveResult<()> {
86        self.walk_mut(hlir);
87
88        if !self.errors.is_empty() {
89            return Err(ResolverErrorKind::ResolverErrors(self.errors.clone()).with_no_span());
90        }
91
92        Ok(())
93    }
94
95    /// Resolve a type from the given identifier path.
96    /// This function attempts to find the type definition in the current [`Namespace`] context.
97    /// If the path is root relative, it will be resolved from the root namespace, otherwise it will be
98    /// resolved relative to the current module path.
99    /// # Returns
100    /// * `Ok(TypeID)` if the type was resolved successfully.
101    /// * `Err(ResolverError)` if the type could not be resolved.
102    /// # Errors
103    /// This function will return an error if the type definition could not be found, describing the failure reason.
104    fn resolve_type(&self, path: &IdentPath) -> ResolveResult<TypeID> {
105        let final_path = path.rebase_from_path(
106            &self
107                .meta
108                .namespace
109                .find_previous_module(self.current_path()),
110        );
111
112        let Some(def) = self.meta.namespace.find_definition(&final_path) else {
113            return Err(resolve_err!(
114                no_span,
115                "Cannot find parent namespace while type checking for path `{}`",
116                path
117            ));
118        };
119
120        if let NamespaceKind::Struct(ty_id) = def.kind {
121            Ok(ty_id)
122        } else {
123            Err(resolve_err!(
124                no_span,
125                "Expected ADT definition at `{}`, but instead found: `{:?}`!",
126                path,
127                def.kind
128            ))
129        }
130    }
131}

133impl TypeResolver<'_> {
134    /// Get a reference to the current identifier path from the top of the path stack.
135    /// This represents the current module or path during the resolution process.
136    #[inline]
137    #[must_use]
138    pub fn current_path(&self) -> &IdentPath {
139        &self.current_path
140    }
141
142    /// Get a mutable reference to the current identifier path from the top of the path stack.
143    /// This represents the current module or path during the resolution process.
144    #[inline]
145    #[must_use]
146    pub fn current_path_mut(&mut self) -> &mut IdentPath {
147        &mut self.current_path
148    }
149
150    /// Push a new segment onto the end of the current identifier path.
151    /// # Example
152    /// If our current path is `::example::inner_module` and we push `MyStruct`, our new path will be
153    /// `::example::inner_module::MyStruct`.
154    #[inline]
155    pub fn push_to_current_path(&mut self, s: &str) {
156        self.current_path_mut().push(s);
157    }
158
159    /// Pop the last segment from the current identifier path.
160    /// # Example
161    /// If our current path is `::example::inner_module::MyStruct` and we pop, our new path will be
162    /// `::example::inner_module`.
163    #[inline]
164    pub fn pop_from_current_path(&mut self) {
165        self.current_path_mut().pop();
166    }
167}

169impl HLIRVisitorMut<'_> for TypeResolver<'_> {
170    fn visit_expr_mut(&mut self, expr: &mut Expr, hlir: &mut HLIRDisjointMut<'_>) {
171        if let ExprKind::StructInit(struct_init) = &mut expr.kind
172            && let RefPath::Unresolved(ty_path) = &struct_init.ty_path
173        {
174            match self.resolve_type(&ty_path.path) {
175                Ok(type_id) => {
176                    struct_init.ty_path =
177                        RefPath::Resolved(ty_path.clone(), ResolvedID::TypeDef(type_id));
178                }
179                Err(e) => self.errors.push(e),
180            }
181        }
182
183        self.super_expr_mut(expr, hlir);
184    }
185
186    fn visit_module_mut(&mut self, module: &mut Module, hlir: &mut HLIRDisjointMut<'_>) {
187        if module.owner_id == OwnerDefId::ROOT_NODE {
188            self.super_module_mut(module, hlir);
189        } else {
190            self.push_to_current_path(module.ident.ident().str());
191            self.super_module_mut(module, hlir);
192            self.pop_from_current_path();
193        }
194    }
195
196    fn visit_function_mut(&mut self, function: &mut Function, hlir: &mut HLIRDisjointMut<'_>) {
197        self.push_to_current_path(function.ident.str());
198        self.super_function_mut(function, hlir);
199
200        for (i, arg) in function.sig.parameters.iter_mut().enumerate() {
201            match arg {
202                Type::Unresolved(Ty::Type(type_path)) => match self.resolve_type(type_path) {
203                    Ok(type_id) => {
204                        *arg = Type::Resolved(KitTy::Abstract(type_id));
205                    }
206                    Err(e) => self.errors.push(e),
207                },
208                Type::Unresolved(Ty::This(sp)) => {
209                    if let Some(impl_ty) = &self.current_impl {
210                        *arg = impl_ty.clone();
211                    } else {
212                        push_resolve_err!(
213                            self,
214                            on_span,
215                            *sp,
216                            "Could not resolve type of `self` function argument in `{}`",
217                            self.current_path()
218                        );
219                    }
220                }
221                _ => {}
222            }
223            if let Type::Unresolved(Ty::Type(type_path)) = arg {
224                if let Ok(type_id) = self.resolve_type(type_path) {
225                    *arg = Type::Resolved(KitTy::Abstract(type_id));
226                } else {
227                    push_resolve_err!(
228                        self,
229                        on_span,
230                        type_path.span,
231                        "Could not resolve type `{}` of function argument `{}` in `{}`",
232                        type_path.path,
233                        function
234                            .sig
235                            .parameter_idents
236                            .get(i)
237                            .cloned()
238                            .unwrap_or_else(|| "??".to_string()),
239                        self.current_path()
240                    );
241                }
242            }
243        }
244
245        if let Type::Unresolved(Ty::Type(type_path)) = &function.sig.output {
246            if let Ok(type_id) = self.resolve_type(type_path) {
247                function.sig.output = Type::Resolved(KitTy::Abstract(type_id));
248            } else {
249                push_resolve_err!(
250                    self,
251                    on_span,
252                    type_path.span,
253                    "Could not resolve type `{}` in function return type of `{}`",
254                    type_path.path,
255                    self.current_path()
256                );
257            }
258        }
259
260        self.pop_from_current_path();
261    }
262
263    fn visit_let_statement_mut(
264        &mut self,
265        id: crate::intermediate::hir::HirId,
266        let_statement: &mut crate::intermediate::hir::nodes::LetStatement,
267        hlir: &mut HLIRDisjointMut<'_>,
268    ) {
269        self.super_let_statement_mut(id, let_statement, hlir);
270        if let Type::Unresolved(Ty::Type(type_path)) = &let_statement.ty {
271            match self.resolve_type(&type_path.path) {
272                Ok(type_id) => {
273                    let_statement.ty = Type::Resolved(KitTy::Abstract(type_id));
274                }
275                Err(e) => {
276                    self.errors.push(e);
277                }
278            }
279        }
280    }
281
282    fn visit_struct_mut(&mut self, structure: &mut Struct, hlir: &mut HLIRDisjointMut<'_>) {
283        self.push_to_current_path(structure.ident.str());
284        let current_struct_path = self.current_path().clone();
285        let current_struct_type = match self.resolve_type(&current_struct_path) {
286            Ok(t) => t,
287            Err(e) => {
288                self.errors.push(e);
289                self.pop_from_current_path();
290                return;
291            }
292        };
293
294        for field_id in &structure.fields {
295            if let Some(field) = hlir.get_hir_node_mut_as::<StructField>(*field_id) {
296                let resolved_type = if let Type::Unresolved(Ty::Type(type_path)) = &field.ty {
297                    match self.resolve_type(&type_path.path) {
298                        Ok(type_id) => Some(type_id),
299                        Err(e) => {
300                            self.errors.push(e);
301                            None
302                        }
303                    }
304                } else {
305                    None
306                };
307
308                if let Some(type_id) = resolved_type {
309                    field.ty = Type::Resolved(KitTy::Abstract(type_id));
310
311                    let struct_info = self
312                        .meta
313                        .type_registry
314                        .get_from_type_id_mut(current_struct_type)
315                        .expect("Has to exist.");
316                    if let Some(field_info) = struct_info.get_field_by_ident_mut(field.ident.str())
317                    {
318                        field_info.ty = Type::Resolved(KitTy::Abstract(type_id));
319                    } else {
320                        push_resolve_err!(
321                            self,
322                            no_span,
323                            "Failed to find struct field `{}` in struct `{}`",
324                            field.ident.string(),
325                            current_struct_path
326                        );
327                    }
328                }
329            }
330        }
331        self.pop_from_current_path();
332    }
333
334    fn visit_impl_mut(&mut self, impl_info: &mut Impl, hlir: &mut HLIRDisjointMut<'_>) {
335        let impl_path = impl_info.self_ty.rebase_from_path(self.current_path());
336
337        if let Some(t) = self.meta.type_registry.find_type_from_path(&impl_path) {
338            self.current_impl = Some(Type::Resolved(t));
339            self.super_impl_mut(impl_info, hlir);
340            self.current_impl = None;
341        } else {
342            push_resolve_err!(
343                self,
344                no_span,
345                "Could not find type for impl at path: `{}`",
346                impl_path
347            );
348        }
349    }
350}

352/// Resolves all types referenced by a path to a [`TypeID`] in the given [`HLIR`].
353/// This function performs a mutable walk over the [`HLIR`], resolving type references.
354/// # Returns
355/// * `Ok(())` if all type references were resolved successfully.
356/// * `Err(ResolverError)` if there were any resolution errors.
357/// # Errors
358/// This function will return an error if any type references could not be resolved.
359/// The returned error may contain multiple resolution errors if there were multiple failures.
360pub fn resolve_types(hlir: &mut HLIR, meta: &mut ProgramMetaData) -> ResolveResult<()> {
361    TypeResolver::new(meta).resolve_types(hlir)
