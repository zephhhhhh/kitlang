1//! The purpose of this module is to build and maintain the namespace scope tree
2//! and resolve associated references (paths to items like types, modules, functions, fields) across the program.
3//! It enforces visibility and local access rules while converting unresolved paths into resolved IDs.
4//!
5//! This pass proceeds in multiple stages:
6//! - Inject builtin types into the root [`Namespace`].
7//! - Walk the [`HLIR`] to construct the [`Namespace`] tree, deferring `impl` items until types and modules exist.
8//! - Visit deferred `impl` items, now that their target types are defined.
9//! - Resolve `use` items by importing target definitions into the current [`Namespace`].
10//! - Do a final mutable pass over expressions to resolve `Path` references.
11//!
12//! Example of an associated reference:
13//! ```ignore
14//! struct Foo;
15//!
16//! impl Foo {
17//!    fn bar() {}
18//! }
19//!
20//! fn main() {
21//!   Foo::bar();
22//! }
23//! ```
24//! In this example, `Foo::bar()` is an associated reference resolved to its owning definition.
25//!
26//! Builtin types are present in the root [`Namespace`] for resolution, though their underlying layout
27//! and type information are defined internally by the compiler rather than as source declarations.
28//!
29//! This module is exclusively focused on global/associated reference resolution and [`Namespace`] building.
30//! Local variable resolution and type resolution are handled by other modules in the resolver.
31

52// Internal mapper used by this module to construct the namespace tree
53// and resolve associated references across modules, types, functions, and uses.
54struct AssociatedReferenceMapper<'a> {
55    pub meta: &'a mut ProgramMetaData,
56
57    /// Tracks all defined global functions, for quick lookup during resolution.
58    /// We check this map first when resolving single-segment paths.
59    pub global_functions: HashMap<String, OwnerDefId>,
60
61    /// Stack of identifier paths representing the current scope during traversal.
62    /// This is used if we need to temporarily change scope while visiting nested items,
63    /// before reverting back to the previous scope.
64    pub path_stack: Vec<IdentPath>,
65
66    /// Lookup table of all `impl` items encountered during the first pass,
67    /// so we can defer visiting them until later.
68    pub impl_path_lut: Vec<(OwnerDefId, IdentPath)>,
69
70    /// Indicates whether the first stage of mapping (building namespaces) is complete.
71    pub stage1_complete: bool,
72
73    /// Collected resolver errors encountered during mapping.
74    pub errors: Vec<ResolverError>,
75    /// Collected resolution failures for unresolved references, to be reported after resolution.
76    /// Each entry contains the [`HirId`] of the item that failed, the [`SpannedIdentPath`] it is trying to access,
77    /// and the [`ResolutionFailure`] reason.
78    pub resolution_failures: Vec<(HirId, SpannedIdentPath, ResolutionFailure)>,
79}

106    /// Executes `f` with `segment` temporarily pushed onto the current path.
107    /// Ensures the segment is popped after `f` completes.
108    #[inline]
109    fn with_pushed_segment<F>(&mut self, segment: &str, f: F)
110    where
111        F: FnOnce(&mut Self),
112    {
113        self.push_to_current_path(segment);
114        f(self);
115        self.pop_from_current_path();
116    }

118    /// Executes `f` with `path` temporarily pushed onto the path stack.
119    /// Ensures the path is popped after `f` completes.
120    #[inline]
121    fn with_pushed_path<F>(&mut self, path: IdentPath, f: F)
122    where
123        F: FnOnce(&mut Self),
124    {
125        self.push_stack(path);
126        f(self);
127        self.pop_stack();

130    /// Injects builtin type definitions into the root [`Namespace`].
131    fn inject_builtin_definitions(&mut self) {
132        fn builtin_namespace(name: &str) -> Namespace {
133            Namespace {
134                ident: name.to_string(),
135                kind: NamespaceKind::Builtin,
136                items: HashMap::new(),
137                id: ResolvedID::OwnerDef(OwnerDefId::ROOT_NODE),
138                vis: Visibility::Public,
139                local: false,
140            }
141        }
142
143        let builtin_types = vec![
144            "i8", "i16", "i32", "i64", "i128", "isize", "u8", "u16", "u32", "u64", "u128", "usize",
145            "f32", "f64", "bool", "string", "char", "()",
146        ];
147
148        builtin_types
149            .into_iter()
150            .map(builtin_namespace)
151            .for_each(|ns| {
152                self.meta.namespace.insert(ns);
153            });
154    }

156    /// Recursively resolves all `use` items in the given [`Namespace`] by
157    /// looking up their target definitions in the `root_namespace` and cloning them
158    /// into the current [`Namespace`].
159    // TODO: This is a bit messy, needs refactoring, also this should reference other items not just clone.
160    fn resolve_uses_in_namespace(
161        root_namespace: &Namespace,
162        namespace: &mut Namespace,
163        current_path: &IdentPath,
164        // so bad...
165        errors: &mut Vec<ResolverError>,
166    ) {
167        let item_idents: Vec<String> = namespace.items.keys().cloned().collect();
168
169        for ident in item_idents {
170            let Some(item_namespace) = namespace.get_mut(&ident) else {
171                continue;
172            };
173            let mut next_path = current_path.clone();
174            next_path.push(&ident);
175            if let NamespaceKind::Use(use_path) = &item_namespace.kind {
176                let target_path = use_path.rebase_from_path(current_path);
177
178                if let Some(target_namespace) = root_namespace.find_definition(&target_path) {
179                    let mut cloned = target_namespace.clone();
180                    cloned.ident.clone_from(&item_namespace.ident);
181                    cloned.vis = item_namespace.vis;
182                    cloned.local = item_namespace.local;
183                    cloned.id = target_namespace.id;
184                    *item_namespace = cloned;
185
186                    Self::resolve_uses_in_namespace(
187                        root_namespace,
188                        item_namespace,
189                        &next_path,
190                        errors,
191                    );
192                } else {
193                    errors.push(resolve_err!(
194                        no_span,
195                        "Failed to resolve use path `{:?}` within namespace `{:?}`",
196                        target_path,
197                        current_path
198                    ));
199                }
200            } else {
201                Self::resolve_uses_in_namespace(root_namespace, item_namespace, &next_path, errors);
202            }
203        }
204    }

206    /// `Entrypoint` function to map all associated references in the given [`HLIR`].
207    /// This will build the [`Namespace`] tree and map, then resolve all associated references,
208    pub fn map_references(&mut self, hlir: &mut HLIR) -> ResolveResult<()> {
209        self.inject_builtin_definitions();
210
211        // Map all except impls..
212        self.visit_root(hlir);
213        self.stage1_complete = true;
214
215        // Manually traverse impl items..
216        let impls = self.impl_path_lut.clone();
217        for (impl_id, impl_path) in impls {
218            if let Some(node) = hlir.owning_node(impl_id)
219                && let Some(impl_info) = node.hir_impl_ref()
220            {
221                self.reset_path_to(impl_path);
222                self.visit_impl(impl_info, hlir);
223            }
224        }
225
226        let root_namespace_clone = self.meta.namespace.clone();
227        let mut resolve_use_errors = Vec::new();
228        Self::resolve_uses_in_namespace(
229            &root_namespace_clone,
230            &mut self.meta.namespace,
231            &IdentPath::new_empty(true),
232            &mut resolve_use_errors,
233        );
234
235        if !resolve_use_errors.is_empty() {
236            self.errors.extend(resolve_use_errors);
237        }
238
239        // Resolve references..
240        self.reset_path();
241        self.walk_mut(hlir);
242
243        if !self.resolution_failures.is_empty() {
244            return Err(
245                ResolverErrorKind::UnresolvedReferences(UnresolvedReferences {
246                    references: self
247                        .resolution_failures
248                        .iter()
249                        .map(|(a, b, c)| UnresolvedReference {
250                            id: *a,
251                            path: b.clone(),
252                            failure: *c,
253                        })
254                        .collect(),
255                })
256                .with_no_span(),
257            );
258        }
259
260        if !self.errors.is_empty() {
261            return Err(ResolverErrorKind::ResolverErrors(self.errors.clone()).with_no_span());
262        }
263
264        Ok(())
265    }

267    /// Finds the last enclosing scope path for the given `path`.
268    /// Where an enclosing scope is something like a module or function that defines a 'border' of accessibility.
269    #[inline]
270    fn find_last_enclosing_path(&self, path: &IdentPath) -> IdentPath {
271        if path.is_empty() {
272            return IdentPath::new_empty(true);
273        }
274        self.meta.namespace.find_previous_enclosing_scope(path)
275    }

277    /// Searches for the definition of `ident_path` starting from `base_path`,
278    #[inline]
279    fn search_backtracking_for_definition(
280        &self,
281        ident_path: &IdentPath,
282        mut base_path: IdentPath,
283    ) -> Option<(IdentPath, ResolvedID)> {
284        let path = ident_path.rebase_from_path(&base_path);
285        if let Some(def) = self.meta.namespace.find_definition(&path) {
286            return Some((path, def.id));
287        }
288
289        let namespace = self.get_namespace(&base_path)?;
290        if namespace.kind == NamespaceKind::Module {
291            None
292        } else {
293            base_path.pop();
294            self.search_backtracking_for_definition(ident_path, base_path)
295        }
296    }
297
298    /// Validates that if the target path is marked as local (private), it is being accessed from within its enclosing scope.
299    /// # Returns
300    /// * `true` if access is valid.
301    /// * `false` if access violates local visibility rules.
302    #[inline]
303    fn validate_local_access(&self, target_path: &IdentPath, base_path: &IdentPath) -> bool {
304        let Some(target_namespace) = self.get_namespace(target_path) else {
305            return false;
306        };
307        if target_namespace.local {
308            let enclosing_path = self.find_last_enclosing_path(target_path);
309            if !base_path.is_subpath_of(&enclosing_path) {
310                return false;
311            }
312        }
313        true
314    }
315
316    /// Searches for the definition of `ident_path` starting from `base_path`.
317    /// This function performs backtracking to find the definition in enclosing scopes,
318    /// while also enforcing visibility and local access rules.
319    /// # Returns
320    /// * `Ok(ResolvedID)` if the definition is found and accessible.
321    /// * `Err(ResolutionFailure)` if the definition is not found or inaccessible.
322    pub fn search_for_definition(
323        &self,
324        ident_path: &IdentPath,
325        base_path: &IdentPath,
326    ) -> Result<ResolvedID, ResolutionFailure> {
327        if ident_path.len() == 1
328            && let Some(id) = self.global_functions.get(ident_path.path_stem())
329        {
330            Ok(ResolvedID::OwnerDef(*id))
331        } else if let Some((p, def)) =
332            self.search_backtracking_for_definition(ident_path, base_path.clone())
333        {
334            if !self.validate_local_access(&p, base_path) {
335                return Err(ResolutionFailure::Inaccessible);
336            }
337
338            let prev_base_path = self.find_last_enclosing_path(base_path);
339
340            let segment_match = p.matching_segment_count(&prev_base_path);
341            let matched_base_path = prev_base_path.subpath(0, segment_match);
342
343            if segment_match >= p.len() {
344                // Fully matched, no need to check further.
345                return Ok(def);
346            }
347
348            let local_check = matched_base_path.extend(&p.segments()[segment_match]);
349            if !self.validate_local_access(&local_check, &matched_base_path) {
350                debug!("{p} Failed local access check");
351                debug!(
352                    "Checked path: `{local_check}` from `{matched_base_path}`  ->  `{ident_path}`"
353                );
354                debug!("Base path: `{base_path}`");
355                return Err(ResolutionFailure::Inaccessible);
356            }
357
358            let inside_func = self
359                .get_namespace(&local_check)
360                .is_some_and(|ns| ns.kind == NamespaceKind::Function);
361
362            if segment_match.saturating_add(1) >= p.len() {
363                // Fully matched, no need to check further.
364                return Ok(def);
365            }
366
367            let len_to_check = p.len().saturating_sub(segment_match);
368
369            for i in 1..len_to_check {
370                let to_check = prev_base_path.extend_path(&IdentPath::from_segments_slice(
371                    &p.segments()[segment_match..=segment_match + i],
372                    true,
373                ));
374
375                let visible = self.get_namespace(&to_check).is_some_and(|n| {
376                    n.vis == Visibility::Public
377                        || (n.kind == NamespaceKind::Function && inside_func)
378                });
379                if !visible {
380                    debug!(
381                        "{p} Failed check at: `{to_check}` from `{base_path}`  ->  `{ident_path}`"
382                    );
383                    return Err(ResolutionFailure::Inaccessible);
384                }
385            }
386
387            Ok(def)
388        } else {
389            Err(ResolutionFailure::NotFound)
390        }
391    }
392}

394impl AssociatedReferenceMapper<'_> {
395    /// Get a reference to the current identifier path from the top of the path stack.
396    #[inline]
397    pub fn current_path(&self) -> &IdentPath {
398        self.path_stack
399            .last()
400            .expect("Always has atleast 1 in stack.")
401    }
402
403    /// Get a mutable reference to the current identifier path from the top of the path stack.
404    #[inline]
405    pub fn current_path_mut(&mut self) -> &mut IdentPath {
406        self.path_stack
407            .last_mut()
408            .expect("Always has atleast 1 in stack.")
409    }
410
411    /// Push a new segment onto the end of the current identifier path.
412    /// # Example
413    /// If our current path is `::example` and we push `inner_module`, our new path will be
414    /// `::example::inner_module`.
415    #[inline]
416    pub fn push_to_current_path(&mut self, s: &str) {
417        self.current_path_mut().push(s);
418    }
419
420    /// Pop the last segment from the end of the current identifier path.
421    /// # Example
422    /// If our current path is `::example::inner_module` and we pop, our new path will be
423    /// `::example`.
424    #[inline]
425    pub fn pop_from_current_path(&mut self) {
426        self.current_path_mut().pop();
427    }
428
429    /// Push a new identifier path onto the top of the path stack.
430    #[inline]
431    pub fn push_stack(&mut self, path: IdentPath) {
432        self.path_stack.push(path);
433    }
434
435    /// Pop the top identifier path from the path stack.
436    #[inline]
437    pub fn pop_stack(&mut self) {
438        self.path_stack.pop();
439    }
440}

442impl AssociatedReferenceMapper<'_> {
443    /// Determines if the given `path` should be marked as local based on its namespace.
444    /// A path is considered local if it is defined within a local scope or is a function
445    #[inline]
446    fn should_be_local(&self, path: &IdentPath) -> bool {
447        let Some(namespace) = self.get_namespace(path) else {
448            return false;
449        };
450        namespace.local || namespace.kind == NamespaceKind::Function
451    }
452
453    /// Get a reference to the [`Namespace`] defined at a given `path`, if it exists.
454    #[inline]
455    fn get_namespace(&self, path: &IdentPath) -> Option<&Namespace> {
456        let mut curr_namespace = &self.meta.namespace;
457        for segment in path.segments() {
458            curr_namespace = curr_namespace.get(segment)?;
459        }
460        Some(curr_namespace)
461    }
462
463    /// Get a mutable reference to the [`Namespace`] defined at a given `path`, if it exists.
464    #[inline]
465    fn get_namespace_mut(&mut self, path: &IdentPath) -> Option<&mut Namespace> {
466        let mut curr_namespace = &mut self.meta.namespace;
467        for segment in path.segments() {
468            curr_namespace = curr_namespace.get_mut(segment)?;
469        }
470        Some(curr_namespace)
471    }
472}

474impl HLIRVisitor for AssociatedReferenceMapper<'_> {
475    fn visit_module(&mut self, module: &Module, hlir: &HLIR) {
476        if module.owner_id == OwnerDefId::ROOT_NODE {
477            self.super_module(module, hlir);
478            return;
479        }
480
481        let current_path = self.current_path().clone();
482        let module_ident = module.ident.ident().string();
483        let local = self.should_be_local(&current_path);
484        if let Some(ns) = self.get_namespace_mut(&current_path) {
485            ns.insert(Namespace {
486                ident: module_ident.clone(),
487                kind: NamespaceKind::Module,
488                items: HashMap::new(),
489                id: ResolvedID::OwnerDef(module.owner_id),
490                vis: module.vis,
491                local,
492            });
493            self.with_pushed_segment(&module_ident, |this| this.super_module(module, hlir));
494        } else {
495            push_resolve_err!(
496                self,
497                on_span,
498                module.ident.span(),
499                "Cannot find parent namespace `{}`",
500                current_path
501            );
502        }
503    }
504
505    fn visit_function(&mut self, function: &Function, hlir: &HLIR) {
506        let current_path = self.current_path().clone();
507        let function_ident = function.ident.string();
508        let local = self.should_be_local(&current_path);
509
510        if function.is_global {
511            self.global_functions
512                .insert(function_ident.clone(), function.owner_id);
513        }
514
515        if self
516            .get_namespace_mut(&current_path)
517            .expect("Namespace path exists")
518            .items
519            .contains_key(&function_ident)
520        {
521            push_resolve_err!(
522                self,
523                on_span,
524                function.ident.span,
525                "Item already defined: `{}`",
526                current_path
527            );
528        }
529        if let Some(ns) = self.get_namespace_mut(&current_path) {
530            ns.items.insert(
531                function_ident.clone(),
532                Namespace {
533                    ident: function_ident.clone(),
534                    kind: NamespaceKind::Function,
535                    items: HashMap::new(),
536                    id: ResolvedID::OwnerDef(function.owner_id),
537                    vis: function.vis,
538                    local,
539                },
540            );
541            self.with_pushed_segment(&function_ident, |this| this.super_function(function, hlir));
542        } else {
543            push_resolve_err!(
544                self,
545                on_span,
546                function.ident.span,
547                "Cannot find parent namespace: `{}`",
548                current_path
549            );
550        }
551    }
552
553    fn visit_struct(&mut self, structure: &Struct, hlir: &HLIR) {
554        fn get_field_info(node: Option<&HirNode>) -> Option<StructField> {
555            if let HirNode::Field(a) = node? {
556                Some(a.clone())
557            } else {
558                None
559            }
560        }
561
562        let current_path = self.current_path().clone();
563        let adt_fields: Option<Vec<StructField>> = structure
564            .fields
565            .iter()
566            .map(|id| hlir.get_hir_node(*id))
567            .map(get_field_info)
568            .collect();
569
570        let struct_type_id = if let Some(resolved_fields) = adt_fields {
571            let fields = resolved_fields
572                .iter()
573                .map(|fi| ADTStructField {
574                    ident: fi.ident.clone(),
575                    ty: fi.ty.clone(),
576                })
577                .collect();
578
579            self.meta
580                .type_registry
581                .register_adt(ADTTypeInfo::new_struct(
582                    structure.owner_id,
583                    self.current_path().clone(),
584                    structure.ident.clone(),
585                    fields,
586                ))
587        } else {
588            push_resolve_err!(
589                self,
590                on_span,
591                structure.ident.span,
592                "Cannot resolve struct fields for `{}`",
593                current_path
594            );
595            0
596        };
597
598        let local = self.should_be_local(&current_path);
599        let struct_ident = structure.ident.string();
600        if let Some(ns) = self.get_namespace_mut(&current_path) {
601            ns.items.insert(
602                struct_ident.clone(),
603                Namespace {
604                    ident: struct_ident,
605                    kind: NamespaceKind::Struct(struct_type_id),
606                    items: HashMap::new(),
607                    id: ResolvedID::OwnerDef(structure.owner_id),
608                    vis: structure.vis,
609                    local,
610                },
611            );
612        } else {
613            push_resolve_err!(
614                self,
615                on_span,
616                structure.ident.span,
617                "Cannot find parent namespace: `{}`",
618                current_path
619            );
620        }
621    }

671    fn visit_impl(&mut self, impl_info: &Impl, hlir: &HLIR) {
672        if self.stage1_complete {
673            // Do the resolve.
674            self.super_impl(impl_info, hlir);
675        } else {
676            let impl_path = impl_info.self_ty.rebase_from_path(self.current_path());
677            self.impl_path_lut.push((impl_info.owner_id, impl_path));
678        }
679    }

681    fn visit_use(&mut self, use_info: &UsePath, hlir: &HLIR) {
682        let current_path = self.current_path().clone();
683
684        for import in &use_info.imports {
685            let import_ident = import.segments().last().expect("Atleast one segment.");
686            if let Some(ns) = self.get_namespace_mut(&current_path) {
687                ns.insert(Namespace {
688                    ident: import_ident.clone(),
689                    kind: NamespaceKind::Use(import.clone()),
690                    items: HashMap::new(),
691                    id: ResolvedID::OwnerDef(OwnerDefId(0)),
692                    vis: use_info.vis,
693                    local: use_info.vis == Visibility::Private,
694                });
695            } else {
696                push_resolve_err!(
697                    self,
698                    on_span,
699                    use_info.span,
700                    "Cannot find parent namespace: `{}`",
701                    current_path
702                );
703            }
704        }
705
706        self.super_use(use_info, hlir);
707    }
708}
709
710impl HLIRVisitorMut<'_> for AssociatedReferenceMapper<'_> {
711    fn visit_module_mut(&mut self, module: &mut Module, hlir: &mut HLIRDisjointMut<'_>) {
712        if module.owner_id == OwnerDefId::ROOT_NODE {
713            self.super_module_mut(module, hlir);
714        } else {
715            self.push_to_current_path(module.ident.ident().str());
716            self.super_module_mut(module, hlir);
717            self.pop_from_current_path();
718        }
719    }
720
721    fn visit_function_mut(&mut self, function: &mut Function, hlir: &mut HLIRDisjointMut<'_>) {
722        self.push_to_current_path(function.ident.str());
723        self.super_function_mut(function, hlir);
724        self.pop_from_current_path();
725    }
726
727    fn visit_path_mut(&mut self, id: HirId, path: &mut RefPath, _hlir: &mut HLIRDisjointMut<'_>) {
728        if !path.is_resolved() {
729            let ident_path = path.ident_path();
730            match self.search_for_definition(ident_path, self.current_path()) {
731                Ok(resolved) => {
732                    path.resolve_to(resolved);
733                }
734                Err(ResolutionFailure::Inaccessible) => {
735                    self.resolution_failures.push((
736                        id,
737                        path.spanned_ident_path().clone(),
738                        ResolutionFailure::Inaccessible,
739                    ));
740                }
741                _ => {}
742            }
743        }
744    }
745
746    fn visit_impl_mut(&mut self, impl_info: &mut Impl, hlir: &mut HLIRDisjointMut<'_>) {
747        let impl_path = impl_info.self_ty.rebase_from_path(self.current_path());
748        self.with_pushed_path(impl_path, |this| this.super_impl_mut(impl_info, hlir));
749    }
750}

753    hlir: &mut HLIR,
754    meta_data: &mut ProgramMetaData,
755) -> ResolveResult<()> {
756    AssociatedReferenceMapper::new(meta_data).map_references(hlir)
