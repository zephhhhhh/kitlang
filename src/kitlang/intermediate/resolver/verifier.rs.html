1//! The purpose of this module is to verify that all reference paths have been resolved after the
2//! resolution phase of the compiler.
3//!
4//! Traverses the [`HLIR`] and records any `RefPath::Unresolved`, and aggregating all such instances into one
5//! `ResolverError` containing all unresolved references.
6//!
7//! This file does not perform resolution or mutate the HLIR.
8//! It only detects unresolved references and surfaces them as diagnostics for the resolution pipeline.
9//!
10//! This module is exclusively focused on verification of unresolved references.
11
12use crate::intermediate::hir::nodes::RefPath;
13use crate::intermediate::hir::visitor::HLIRVisitor;
14use crate::intermediate::hir::{HLIR, HirId};
15
16use crate::intermediate::resolver::errors::{
17    ResolutionFailure, ResolveResult, ResolverErrorKind, UnresolvedReference, UnresolvedReferences,
18};
19
20/// Visitor that checks for unresolved references in the HLIR.
21/// It collects all unresolved references found during the visit.
22#[derive(Default)]
23struct UnresolvedReferenceChecker {
24    /// All found unresolved references.
25    pub unresolved_references: Vec<UnresolvedReference>,
26}
27
28impl HLIRVisitor for UnresolvedReferenceChecker {
29    fn visit_path(&mut self, id: HirId, path: &RefPath, hlir: &HLIR) {
30        if let RefPath::Unresolved(ident_path) = path {
31            self.unresolved_references.push(UnresolvedReference {
32                path: ident_path.clone(),
33                id,
34                failure: ResolutionFailure::NotFound,
35            });
36        }
37
38        self.super_path(id, path, hlir);
39    }
40}
41
42impl UnresolvedReferenceChecker {
43    /// Verifies all references in the given HLIR.
44    /// Visits all `path` nodes and collects any that are still unresolved.
45    /// # Returns
46    /// * `Ok(())` if all references are resolved.
47    /// * `Err(ResolverError)` if there are any unresolved references.
48    /// # Errors
49    /// This function will return an error if there are any unresolved references in the [`HLIR`] remaining.
50    /// The returned error will contain all unresolved references found.
51    pub fn verify_references(mut self, hlir: &HLIR) -> ResolveResult<()> {
52        self.visit_root(hlir);
53
54        if self.unresolved_references.is_empty() {
55            Ok(())
56        } else {
57            Err(
58                ResolverErrorKind::UnresolvedReferences(UnresolvedReferences {
59                    references: self.unresolved_references,
60                })
61                .with_no_span(),
62            )
63        }
64    }
65}
66
67/// Verifies all references in the given HLIR.
68/// Visits all `path` nodes and collects any that are still unresolved.
69/// # Returns
70/// * `Ok(())` if all references are resolved.
71/// * `Err(ResolverError)` if there are any unresolved references.
72/// # Errors
73/// This function will return an error if there are any unresolved references in the [`HLIR`] remaining.
74/// The returned error will contain all unresolved references found.
75pub fn verify_references(hlir: &HLIR) -> ResolveResult<()> {
76    UnresolvedReferenceChecker::default().verify_references(hlir)
77}
